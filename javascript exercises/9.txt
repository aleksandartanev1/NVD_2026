The function main() below contains a small program. Hoisting works inside a function exactly as in the global code: every call of main() creates a new execution context, which goes through its own creation phase.

function main() {
  console.log(describe(item));
  console.log(format(price));
  console.log(total);
  console.log(format(discount(total)));

  var total = price * qty;
  const format = function (amount) {
    return amount.toFixed(2) + ' ' + currency;
  };
  let price = 9.5;
  var item = 'Book';
  var discount = function (amount) {
    return amount > limit ? amount - amount * rate / 100 : amount;
  };
  const qty = 2;
  let currency = 'EUR';
  var limit = 15;
  const rate = 10;

  function describe(name) {
    return 'Item: ' + name + ' x' + qty;
  }
}
